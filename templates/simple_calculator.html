import streamlit as st # Set up the web page title st.title("🔢 Simple Calculator") st.write("Enter your numbers and choose an operation.") # Create two input boxes for numbers num1 = st.number_input("Enter first number", value=0.0) num2 = st.number_input("Enter second number", value=0.0) # Create a dropdown menu for selecting the mathematical operation operation = st.selectbox("Choose an operation", ("Add", "Subtract", "Multiply", "Divide")) # Perform the calculation based on the selection result = 0 if st.button("Calculate"): if operation == "Add": result = num1 + num2 st.success(f"**Result:** {num1} + {num2} = {result}") elif operation == "Subtract": result = num1 - num2 st.success(f"**Result:** {num1} - {num2} = {result}") elif operation == "Multiply": result = num1 * num2 st.success(f"**Result:** {num1} × {num2} = {result}") elif operation == "Divide": if num2 == 0: st.error("Error: Cannot divide by zero!") else: result = num1 / num2 st.success(f"**Result:** {num1} ÷ {num2} = {result}")
